create table WSTREQDETAIL
(
	NFRMNO	NUMBER(3,0),
	VORGNO	VARCHAR2(6),
	CYEAR	CHAR(2),
	CYEAR2	CHAR(4),
	NRUNNO	NUMBER(6,0),
	WSTNO	varchar2(2),
	QTY	NUMBER(2,0),
	primary key(NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO,WSTNO),
	foreign key(NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO) references FORM(NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO),
	foreign key(WSTNO) references WSTTYPEMST(WSTNO)
);
