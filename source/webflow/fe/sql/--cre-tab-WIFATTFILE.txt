create table WIFATTFILE
(
	NFRMNO	NUMBER(3,0),
	VORGNO	VARCHAR2(6),
	CYEAR	CHAR(2),
	CYEAR2	CHAR(4),
	NRUNNO	NUMBER(6,0),
	TYPENO	CHAR(2),
	ITEMNO	NUMBER(2,0),
	SFILE	VARCHAR2(70),
	primary key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO,TYPENO,ITEMNO),
	foreign key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO) references FORM(NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO)
);
