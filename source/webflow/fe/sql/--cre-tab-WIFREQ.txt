create table WIFREQ
(
	NFRMNO	NUMBER(3,0),
	VORGNO	VARCHAR2(6),
	CYEAR	CHAR(2),
	CYEAR2	CHAR(4),
	NRUNNO	NUMBER(6,0),
	SUBJECT	VARCHAR2(20),
	PLACE	VARCHAR2(20),
	SECRESPONSIBLE	VARCHAR2(6),
	--DEPTRESPONSIBLE	VARCHAR2(6),
	TYPEWORKER	CHAR(1),
	STARTDATE	DATE,
	ENDDATE		DATE,	
	STARTTIME	CHAR(5),
	ENDTIME		CHAR(5),
	HOTWKSTATUS	CHAR(1),
	CONFINESTATUS	CHAR(1),
	HIGHPOSSTATUS	CHAR(1),
	CHEMSTATUS	CHAR(1),
	GASUSESTATUS	CHAR(1),
	CRANESTATUS	CHAR(1),
	FORKLIFTSTATUS	CHAR(1),
	ENVINSPECT	CHAR(1),
	STINSPECT	CHAR(1),
	primary key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO),
	foreign key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO) references FORM(NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO),
	foreign key (SECRESPONSIBLE) references AMEC.PSECTION(SSECCODE)
);
