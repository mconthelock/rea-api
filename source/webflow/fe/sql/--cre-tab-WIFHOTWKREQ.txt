create table WIFHOTWKREQ
(
	NFRMNO	NUMBER(3,0),
	VORGNO	VARCHAR2(6),
	CYEAR	CHAR(2),
	CYEAR2	CHAR(4),
	NRUNNO	NUMBER(6,0),
	HOTWKNO	CHAR(1),
	primary key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO),
	foreign key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO) references FORM(NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO),
	foreign key (HOTWKNO) references WIFHOTWKMST(HOTWKNO)
);
