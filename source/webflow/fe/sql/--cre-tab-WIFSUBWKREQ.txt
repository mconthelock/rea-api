create table WIFSUBWKREQ
(
	NFRMNO	NUMBER(3,0),
	VORGNO	VARCHAR2(6),
	CYEAR	CHAR(2),
	CYEAR2	CHAR(4),
	NRUNNO	NUMBER(6,0),
	SREGCYEAR	CHAR(4),
	COMPANYID	NUMBER(4,0),
	WORKERID	NUMBER(2,0),
	primary key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO,WORKERID),
	foreign key (NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO) references FORM(NFRMNO,VORGNO,CYEAR,CYEAR2,NRUNNO),
	foreign key (SREGCYEAR,COMPANYID) references WIFSUPCONMST(CYEAR,COMPANYID)
);
