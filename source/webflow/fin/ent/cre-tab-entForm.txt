create table rqtEntForm(
	nFrmNo 		number(3, 0),
	vOrgNo 		varchar2(5) default '00',
	cYear 		char(2),
	cYear2 		char(4),
	nRunNo 		number(6, 0),
	payto		varchar2(256),	
	purpose 	varchar2(256),
	paydate		date,
	place		varchar2(256),
	description	varchar2(256),
	primary key (nFrmNo, vOrgNo, cYear, cYear2, nRunNo),
	foreign key (nFrmNo, vOrgNo, cYear, cYear2, nRunNo) references form(nFrmNo, vOrgNo, cYear, cYear2, nRunNo)
);
