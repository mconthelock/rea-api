select 'EPL'||substr(rl.cyear2,3,2)||'-'||lpad(rl.nrunno, 6, '0') as REQNO , rl.DWGNO , rl.REVNO , rl.EPLREVNO  , rl.VENDORNAME , r.DWGTYPE , r.SENDBY From form f, REQDRAW r, REQDRAWLIST rl where f.nfrmno = r.nfrmno and f.vorgno = r.vorgno and f.cyear = r.cyear and f.cyear2 = r.cyear2 and f.nrunno = r.nrunno and r.nfrmno = rl.nfrmno and r.vorgno = rl.vorgno and r.cyear = rl.cyear and r.cyear2 = rl.cyear2 and r.nrunno = rl.nrunno and r.type = 'Q' and f.cst in ('1','2') and rl.cyear2||rl.nrunno||rl.ID not in (select rn.qcyear2||rn.nrunno||rn.QID from RNDRAW rn, form f where rn.nfrmno = f.nfrmno and rn.vorgno = f.vorgno and rn.cyear = f.cyear and rn.cyear2 = f.cyear2 and rn.nrunno = f.nrunno and f.cst in ('1','2'))
union
select REQNO , DWGNO , REVNO , '' as EPLREVNO , VENDORNAME , DWGTYPE , SENDBY from msdraw where status = 'I'
union
select 'EPL'||substr(rl.cyear2,3,2)||'-'||lpad(rl.nrunno, 6, '0') as REQNO , rl.DWGNO , rl.REVNO  , rl.EPLREVNO , rl.VENDORNAME , r.DWGTYPE , r.SENDBY From form f, REQDRAW r, REQDRAWLIST rl where f.nfrmno = r.nfrmno and f.vorgno = r.vorgno and f.cyear = r.cyear and f.cyear2 = r.cyear2 and f.nrunno = r.nrunno and r.nfrmno = rl.nfrmno and r.vorgno = rl.vorgno and r.cyear = rl.cyear and r.cyear2 = rl.cyear2 and r.nrunno = rl.nrunno and r.type in ('V','S','J') and f.cst in ('1')
)


CREATE VIEW V_QREQDRAW AS (
select a.* , (a.eplapv + 150) as returndate From (
select r.type , r.dwgtype , r.sendby , rl.* , a.sname , sec.ssec , f.cst , f.dreqdate  , (select dapvdate from flow where nfrmno = r.nfrmno and vorgno = r.vorgno and cyear = r.cyear and cyear2 = r.cyear2 and nrunno = r.nrunno and cextdata = '01') as demreqapv , (select dapvdate from flow where nfrmno = r.nfrmno and vorgno = r.vorgno and cyear = r.cyear and cyear2 = r.cyear2 and nrunno = r.nrunno and cextdata = '02') as eplapv , (select dapvdate from flow where nfrmno = r.nfrmno and vorgno = r.vorgno and cyear = r.cyear and cyear2 = r.cyear2 and nrunno = r.nrunno and cextdata = '03') as reqreceive From reqdraw r, reqdrawlist rl , form f , amec.aemployee a , amec.psection sec  where r.nfrmno = rl.nfrmno and r.vorgno = rl.vorgno and r.cyear = rl.cyear and r.cyear2 = rl.cyear2 and r.nrunno = rl.nrunno and r.type = 'Q' and f.nfrmno = r.nfrmno and f.vorgno = r.vorgno and f.cyear = r.cyear and f.cyear2 = r.cyear2 and f.nrunno = r.nrunno and f.cst in ('2','1') and f.vreqno = a.sempno and a.sseccode = sec.sseccode
) a
)

 select rn.* , f.vreqno , f.vinputer , f.vremark , f.dreqdate , f.creqtime , f.cst  from rndraw rn, form f where rn.nfrmno = f.nfrmno and rn.vorgno = f.vorgno and rn.cyear = f.cyear and rn.cyear2 = f.cyear2 and rn.nrunno = f.nrunno

CREATE VIEW V_RQDRAW AS (
select a.nfrmno , a.vorgno , a.EPLNOTE , a.cyear , a.cyear2 , a.NRUNNO , a.ID , a.DEMREQAPV , a.REVNO , a.REASON , a.EPLREVNO , a.SSEC , a.SNAME , a.EPLAPV , a.REQRECEIVE, a.returndate ,a.cst,'EPL'||SUBSTR(a.CYEAR2,3,2)||'-'||lpad(a.NRUNNO,6, '0') as REQNO , a.DREQDATE , a.DWGNO , a.PARTNAME , a.ITEMNO , a.VENDORNAME , a.QTY , a.SSEC||' / '||a.SNAME as REQTER , CASE WHEN a.EPLREVNO is NULL THEN a.REVNO ELSE a.EPLREVNO END as REV , 
 CASE 
 WHEN a.STATUS = 'R' THEN 'Finish' 
 WHEN a.STATUS = 'N' THEN 'NON-Return' 
 WHEN a.STATUS is NULL THEN
 CASE WHEN RETURNDATE is NULL THEN '' 
 WHEN RETURNDATE < SYSDATE THEN 'Remain'
 ELSE 'Inuse' END
 END as STSTATUS from (
 select v.* , (select type from V_RNDRAW where qcyear2 = v.cyear2 and qnrunno = v.nrunno and qid = v.id and cst in ('1','2')) status From V_QREQDRAW v 
 ) a 
)

CREATE VIEW V_RQDRAW AS(
select a.nfrmno , a.vorgno , a.EPLNOTE , a.cyear , a.cyear2 , a.NRUNNO , a.ID , a.DEMREQAPV , a.REVNO , a.REASON , a.EPLREVNO , a.SSEC , a.SNAME , a.EPLAPV ,  a.REQRECEIVE, a.returndate  ,a.cst,'EPL'||SUBSTR(a.CYEAR2,3,2)||'-'||lpad(a.NRUNNO,6, '0') as REQNO  , a.DREQDATE , a.DWGNO , a.PARTNAME , a.ITEMNO , a.VENDORNAME , a.QTY , a.SSEC||' / '||a.SNAME as REQTER  , CASE WHEN a.EPLREVNO is NULL THEN a.REVNO ELSE a.EPLREVNO END as REV , 
			   CASE 
				  WHEN a.STATUS = 'R' THEN 'Finish' 
				  WHEN a.STATUS = 'N' THEN 'NON-Return' 
				  WHEN a.STATUS is NULL  THEN
					CASE WHEN RETURNDATE is NULL THEN '' 
					     WHEN RETURNDATE < SYSDATE THEN 'Remain'
						 ELSE 'Inuse' END
			   END as STSTATUS   from (
			select v.* , (select type from V_RNDRAW where qcyear2 = v.cyear2 and qnrunno = v.nrunno and qid = v.id  and cst in ('1','2')) status From V_QREQDRAW v 
	) a 
)

select * From (
select * From V_RQDRAW v LEFT JOIN (select h.coverno , h.cdate as createcover , l.cyear2 as ccyear2 , l.nrunno as cnrunno , l.ctype , l.RCYEAR2 , l.rnrunno , l.rid from  COVERQH h , COVERQL l where h.cyear2 = l.cyear2 and h.nrunno = l.nrunno and h.ctype = l.ctype) c
ON V.cyear2 = c.rcyear2 and V.nrunno = c.rnrunno and V.ID = c.RID
)rq LEFT JOIN (select  rn.QID, rn.qnrunno , rn.qcyear2 ,
	CASE  
		WHEN rn.type = 'N' THEN 'NR'||SUBSTR(rn.CYEAR2,3,2)||'-'||lpad(rn.NRUNNO,6, '0')
		WHEN rn.type = 'R' THEN 'RE'||SUBSTR(rn.CYEAR2,3,2)||'-'||lpad(rn.NRUNNO,6, '0')
	END as returnno 

, f.dreqdate as rnreqdate , fl.DAPVDATE as rnapvdate From rndraw rn, form f, flow fl where rn.nfrmno = f.nfrmno and rn.vorgno =  f.vorgno and rn.cyear = f.cyear and rn.cyear2 = f.cyear2 and rn.nrunno = f.nrunno and f.nfrmno = fl.nfrmno and f.vorgno = fl.vorgno and f.cyear2 = fl.cyear2 and f.nrunno = fl.nrunno and fl.cextdata = '03'

)rn on rq.cyear2 = rn.qcyear2 and rq.nrunno = rn.qnrunno and rq.ID = rn.QID
